# The avatar library for xsdk titles: libxsdk_avatar.a, built with xsdk's toolchain (build.sh runs it). A title adds
# this folder and links the library:
#   add_subdirectory(<this folder> xsdk_avatar)
#   target_link_libraries(<title> PRIVATE xsdk_avatar)
cmake_minimum_required(VERSION 3.25)
project(xsdk_avatar C CXX ASM)

if(NOT XSDK)
	message(FATAL_ERROR "Configure with xsdk's toolchain: -DCMAKE_TOOLCHAIN_FILE=<xsdk>/cmake/xsdk.cmake")
endif()

# The shaders: GLSL through xost's glslang and compiler (xsdk's README; glslang in xenos-toolchain/tools/glslang),
# with xsdk's build/host/libxsc.so built first.
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(
	OUTPUT "${generated}/avatar_shaders.h"
	COMMAND ${CMAKE_COMMAND} -E make_directory "${generated}"
	COMMAND sh -c "python3 '${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_shaders.py' '${XSDK_ROOT}' \
'${CMAKE_CURRENT_BINARY_DIR}/shader_work' > '${generated}/avatar_shaders.h.new'"
	COMMAND ${CMAKE_COMMAND} -E rename "${generated}/avatar_shaders.h.new" "${generated}/avatar_shaders.h"
	DEPENDS tools/gen_shaders.py
	VERBATIM)

# The library's XAM imports, as thunks generated by xsdk's tool.
file(STRINGS src/imports.txt import_lines REGEX "^[^#].*[^ ]")
set(import_sources)
foreach(line IN LISTS import_lines)
	string(REGEX REPLACE "^[^ ]+ +[^ ]+ +([^ ]+).*" "\\1" name "${line}")
	list(APPEND import_sources "${CMAKE_CURRENT_BINARY_DIR}/imports/${name}.S")
endforeach()
add_custom_command(
	OUTPUT ${import_sources}
	COMMAND python3 "${XSDK_ROOT}/tools/gen_imports.py" "${CMAKE_CURRENT_SOURCE_DIR}/src/imports.txt"
		"${CMAKE_CURRENT_BINARY_DIR}/imports"
	DEPENDS src/imports.txt "${XSDK_ROOT}/tools/gen_imports.py"
	VERBATIM)

file(GLOB library_sources CONFIGURE_DEPENDS src/*.c)
add_library(xsdk_avatar STATIC ${library_sources} ${import_sources} "${generated}/avatar_shaders.h")
target_include_directories(xsdk_avatar PUBLIC include PRIVATE src "${generated}")
target_compile_options(xsdk_avatar PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu17 -O2 -Wall -Wextra -Werror>)

if(PROJECT_IS_TOP_LEVEL)
	add_subdirectory(samples/viewer)
	add_subdirectory(samples/hello)
	add_subdirectory(samples/store)
endif()
